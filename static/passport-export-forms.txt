/*
วิธีส่งออก (ผู้มีสิทธิ์แก้ไข Google Forms)
1. เปิด script.google.com ด้วยบัญชีที่แก้ไขฟอร์มได้ สร้าง New project
2. คัดลอกข้อความทั้งหมดในไฟล์นี้ลง Code.gs แล้วกด Save
3. เลือกฟังก์ชัน exportPassportForms แล้วกด Run อนุญาตให้อ่าน Forms และสร้างไฟล์ใน Drive
4. ดาวน์โหลด YFSC-Forms-Import.json จาก My Drive จากนั้นเข้าเมนู Training Passport > หลักสูตรและแบบทดสอบ > นำเข้าข้อสอบ Google Forms ใน Cloudflare
หากระบบระบุว่าหาฟอร์มไม่พบ ให้ใส่ลิงก์ /edit ในช่อง editUrl ของหลักสูตรนั้น แล้วรันอีกครั้ง
สคริปต์อ่านคำถามจากต้นฉบับ สร้างไฟล์ JSON ใน Drive ของผู้รัน ไฟล์มีเฉลย ให้เฉพาะผู้ดูแลใช้
*/
function videoId(text){
  const m=String(text||'').match(/(?:https?:\/\/)?(?:www\.|m\.)?(?:youtube(?:-nocookie)?\.com\/(?:watch\?[^\s]*?v=|embed\/|shorts\/)|youtu\.be\/)([\w-]{11})(?![\w-])/i);
  return m?m[1]:null;
}
function formVideos(form){
  const videos={};
  try{
    const response=UrlFetchApp.fetch('https://forms.googleapis.com/v1/forms/'+encodeURIComponent(form.getId()),{
      headers:{Authorization:'Bearer '+ScriptApp.getOAuthToken()},muteHttpExceptions:true});
    if(response.getResponseCode()===200){
      for(const item of JSON.parse(response.getContentText()).items||[]){
        const id=videoId(item.videoItem?.video?.youtubeUri);
        if(id)videos[parseInt(item.itemId,16)>>>0]=id;
      }
    }
  }catch(e){}
  return videos;
}
function publishedVideos(form){
  const videos={};
  try{
    const html=UrlFetchApp.fetch(form.getPublishedUrl(),{muteHttpExceptions:true}).getContentText();
    const data=html.match(/FB_PUBLIC_LOAD_DATA_\s*=\s*(\[[\s\S]*?\])\s*;\s*<\/script>/);
    if(!data)return videos;
    const items=JSON.parse(data[1])[1][1];
    for(const item of items){
      if(item[3]!==12)continue;
      const ids=new Set();
      function scan(v){
        if(Array.isArray(v))v.forEach(scan);
        else if(typeof v==='string'){
          const id=videoId(v)||(/^[\w-]{11}$/.test(v)?v:null);
          if(id)ids.add(id);
        }
      }
      item.slice(4).forEach(scan);
      if(ids.size===1)videos[item[0]>>>0]=[...ids][0];
    }
  }catch(e){}
  return videos;
}
function exportPassportForms(){
  const out=[],errors=[];
  let owned;
  function openCourse(c){
    if(c.editUrl)return FormApp.openByUrl(c.editUrl);
    const resp=UrlFetchApp.fetch(c.url,{followRedirects:false,muteHttpExceptions:true});
    const headers=resp.getAllHeaders(),url=headers.Location||headers.location;
    if(!url)throw Error('กรุณาใส่ลิงก์แก้ไข /edit');
    const canonical=s=>String(s).split('?')[0].replace(/\/$/,'');
    try{return FormApp.openByUrl(url);}catch(e){}
    if(!owned){
      owned={};
      const files=DriveApp.getFilesByType(MimeType.GOOGLE_FORMS);
      while(files.hasNext()){
        const file=files.next();
        try{const form=FormApp.openById(file.getId());owned[canonical(form.getPublishedUrl())]=file.getId();}catch(e){}
      }
    }
    const formId=owned[canonical(url)];
    if(!formId)throw Error('บัญชีนี้ต้องมีสิทธิ์แก้ไขฟอร์ม หรือใส่ลิงก์ /edit');
    return FormApp.openById(formId);
  }
  for(const c of PASSPORT_FORMS){
    try{
      const f=openCourse(c), questions=[], items=f.getItems();
      const breaks=items.filter(i=>String(i.getType())==='PAGE_BREAK');
      const pages=[{title:'',next:null},...breaks.map(i=>({title:i.getTitle(),next:null}))];
      const indexes=new Map(breaks.map((i,n)=>[i.getId(),n+1]));
      function destination(item,page){
        const nav=item.getPageNavigationType && item.getPageNavigationType();
        if(!nav||String(nav)==='CONTINUE')return page+1<pages.length?page+1:-1;
        if(String(nav)==='SUBMIT')return -1;
        const dest=String(nav)==='GO_TO_PAGE' && (item.getGoToPage?item.getGoToPage():item.getGotoPage());
        const index=dest?indexes.get(dest.getId()):null;
        if(!Number.isInteger(index)||index<=page)throw Error('พบเส้นทางย้อนกลับในส่วน '+(page+1)+' กรุณาตรวจเส้นทางในฟอร์ม');
        return index;
      }
      let page=0,apiVideos,publicVideos;
      const videos=[];
      pages[pages.length-1].next=-1;
      for(const item of items){
        const type=String(item.getType());
        if(type==='PAGE_BREAK'){pages[page].next=destination(item.asPageBreakItem(),page);page++;continue;}
        if(type==='VIDEO'){
          const video=item.asVideoItem(), key=item.getId()>>>0;
          if(!apiVideos)apiVideos=formVideos(f);
          let id=apiVideos[key];
          if(!id){
            if(!publicVideos)publicVideos=publishedVideos(f);
            id=publicVideos[key]||videoId(item.getTitle())||videoId(video.getHelpText());
          }
          if(!id)throw Error('โปรดใส่ลิงก์ YouTube ของคลิปในคำอธิบายรายการวิดีโอ “'+(item.getTitle()||c.topic)+'” แล้วส่งออกอีกครั้ง');
          videos.push({page,before:questions.length,id,title:item.getTitle()||c.topic});
          continue;
        }
        if(type==='SECTION_HEADER')continue;
        const access={MULTIPLE_CHOICE:['single','asMultipleChoiceItem'],LIST:['single','asListItem'],CHECKBOX:['checkbox','asCheckboxItem'],TEXT:['text','asTextItem'],PARAGRAPH_TEXT:['paragraph','asParagraphTextItem']}[type];
        if(!access)throw Error('ตรวจสอบรูปแบบคำถาม '+type+' ใน '+item.getTitle());
        const q=item[access[1]](),choices=q.getChoices?q.getChoices():[];
        const other=!!(q.hasOtherOption&&q.hasOtherOption());
        const routes=choices.map(x=>x.getPageNavigationType && x.getPageNavigationType()?destination(x,page):null);
        let points=0;if(f.isQuiz()&&q.getPoints)points=q.getPoints()||0;
        const correct=[];
        choices.forEach((x,i)=>{if(f.isQuiz()&&x.isCorrectAnswer&&x.isCorrectAnswer())correct.push(i);});
        questions.push({page,other,routes,title:item.getTitle(),kind:access[0],choices:choices.map(x=>x.getValue()),correct,required:q.isRequired(),points});
      }
      if(!questions.length)throw Error('ไม่มีข้อคำถาม');
      out.push({id:c.id,topic:c.topic,type:'Training',document:c.id,revision:'',pass:c.passScore,pages,videos,questions});
    }catch(err){errors.push(c.id+' '+c.topic+' — '+err.message);}
  }
  if(errors.length)console.log(errors.join('\n'));
  if(!out.length)throw Error('ยังสร้างไฟล์ไม่ได้ เพราะส่งออกได้ 0 หลักสูตร ดูสาเหตุรายหลักสูตรใน Execution log ด้านบน');
  const file=DriveApp.createFile('YFSC-Forms-Import.json',JSON.stringify({format:'yfsc-forms-v3',courses:out},null,2),MimeType.PLAIN_TEXT);
  console.log('ส่งออกได้ '+out.length+' จาก '+PASSPORT_FORMS.length+' หลักสูตร');
  console.log('เปิดไฟล์ YFSC-Forms-Import.json ที่นี่ '+file.getUrl());
}
